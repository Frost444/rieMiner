cmake_minimum_required(VERSION 3.24)
# CMake's generated CUDA compiler file requires slash-normalized Windows paths.
# Accept an explicitly supplied nvcc.exe path containing spaces/backslashes.
if(DEFINED CMAKE_CUDA_COMPILER)
  file(TO_CMAKE_PATH "${CMAKE_CUDA_COMPILER}" normalized_nvcc)
  set(CMAKE_CUDA_COMPILER "${normalized_nvcc}" CACHE FILEPATH "CUDA compiler" FORCE)
endif()
# Never use the build machine's device ordinal or UUID as a source default.
# Native SASS plus forward-JIT PTX reproduces the qualified 8.6 minimum.
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES "86-real;86-virtual" CACHE STRING "CUDA architectures")
endif()
project(RiecoinSilverR346 LANGUAGES CXX)
include(CTest)
option(R322_BUILD_CUDA "Build the R322 GPU engine" ON)
set(GMP_ROOT "" CACHE PATH "GMP prefix: include/gmp.h and lib/gmp.lib (Windows) or libgmp")
if(WIN32 AND NOT GMP_ROOT)
  set(GMP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/windows-deps")
endif()
find_path(GMP_INCLUDE_DIR gmp.h HINTS "${GMP_ROOT}/include" REQUIRED)
find_library(GMP_LIBRARY NAMES gmp libgmp HINTS "${GMP_ROOT}/lib" REQUIRED)
add_library(r322_gmp INTERFACE)
target_include_directories(r322_gmp INTERFACE "${GMP_INCLUDE_DIR}")
target_link_libraries(r322_gmp INTERFACE "${GMP_LIBRARY}")
add_executable(r322-bounds-test r322-bounds-test.cpp)
target_compile_features(r322-bounds-test PRIVATE cxx_std_17)
target_link_libraries(r322-bounds-test PRIVATE r322_gmp)
add_test(NAME r322-bounds COMMAND r322-bounds-test)
add_executable(r322-direct-basis-test r322-direct-basis-test.cpp)
target_compile_features(r322-direct-basis-test PRIVATE cxx_std_17)
target_include_directories(r322-direct-basis-test PRIVATE r312-extra r184-deps)
target_link_libraries(r322-direct-basis-test PRIVATE r322_gmp)
if(WIN32)
  target_link_libraries(r322-direct-basis-test PRIVATE bcrypt)
else()
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  target_link_libraries(r322-direct-basis-test PRIVATE OpenSSL::Crypto)
endif()
add_test(NAME r322-direct-basis COMMAND r322-direct-basis-test)
add_executable(r346-canonical-bridge-test r346-canonical-bridge-test.cpp)
target_compile_features(r346-canonical-bridge-test PRIVATE cxx_std_17)
target_include_directories(r346-canonical-bridge-test PRIVATE r312-extra r184-deps)
target_link_libraries(r346-canonical-bridge-test PRIVATE r322_gmp)
if(WIN32)
  target_link_libraries(r346-canonical-bridge-test PRIVATE bcrypt)
else()
  target_link_libraries(r346-canonical-bridge-test PRIVATE OpenSSL::Crypto)
endif()
add_test(NAME r346-canonical-bridge COMMAND r346-canonical-bridge-test)
if(WIN32 AND EXISTS "${GMP_ROOT}/bin/libgmp-10.dll")
  add_custom_command(TARGET r322-bounds-test POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${GMP_ROOT}/bin/libgmp-10.dll" "$<TARGET_FILE_DIR:r322-bounds-test>")
endif()
if(NOT R322_BUILD_CUDA)
  return()
endif()
enable_language(CUDA)
find_package(CUDAToolkit REQUIRED)
find_package(Threads REQUIRED)
add_executable(r346-canonical-gpu-test EXCLUDE_FROM_ALL r346-canonical-gpu-test.cu)
set_target_properties(r346-canonical-gpu-test PROPERTIES CUDA_STANDARD 17 CXX_STANDARD 17)
target_include_directories(r346-canonical-gpu-test PRIVATE r312-extra r184-deps r178-deps)
target_link_libraries(r346-canonical-gpu-test PRIVATE r322_gmp CUDA::cudart_static)
if(WIN32)
  target_link_libraries(r346-canonical-gpu-test PRIVATE bcrypt)
else()
  target_link_libraries(r346-canonical-gpu-test PRIVATE OpenSSL::Crypto)
endif()
# Exclude the Windows-specific GMP ABI header from the inherited include tree.
# The dependency selected above is authoritative on each target OS.
file(GLOB inherited_headers CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/r178-deps/*")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/r178-headers")
foreach(header IN LISTS inherited_headers)
  get_filename_component(name "${header}" NAME)
  if(NOT IS_DIRECTORY "${header}" AND NOT name STREQUAL "gmp.h")
    configure_file("${header}" "${CMAKE_CURRENT_BINARY_DIR}/r178-headers/${name}" COPYONLY)
  endif()
endforeach()
add_executable(riecoin-r346
  r346-source/r251_miner.cu r346-source/r259_executor.cpp
  r289-sos.cu r294-sos.cu r297-sos.cu)
set_target_properties(riecoin-r346 PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
target_include_directories(riecoin-r346 PRIVATE
  r346-source
  r312-extra "${CMAKE_CURRENT_SOURCE_DIR}" r184-deps
  "${CMAKE_CURRENT_BINARY_DIR}/r178-headers"
  a54-source a19-wizz-closure-20260907 r281-cgbn)
target_compile_options(riecoin-r346 PRIVATE
  "$<$<COMPILE_LANGUAGE:CUDA>:--default-stream=per-thread>"
  "$<$<AND:$<COMPILE_LANGUAGE:CUDA>,$<CONFIG:Release>>:-O3>")
if(MSVC)
  target_compile_options(riecoin-r346 PRIVATE
    "$<$<COMPILE_LANGUAGE:CXX>:/EHsc;/Zc:preprocessor>"
    "$<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/EHsc,/Zc:preprocessor>")
  target_link_libraries(riecoin-r346 PRIVATE bcrypt)
endif()
target_link_libraries(riecoin-r346 PRIVATE r322_gmp CUDA::cudart_static CUDA::cuda_driver Threads::Threads)
if(NOT WIN32)
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  target_link_libraries(riecoin-r346 PRIVATE OpenSSL::Crypto)
endif()
if(WIN32 AND EXISTS "${GMP_ROOT}/bin/libgmp-10.dll")
  add_custom_command(TARGET riecoin-r346 POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${GMP_ROOT}/bin/libgmp-10.dll" "$<TARGET_FILE_DIR:riecoin-r346>")
endif()
install(TARGETS riecoin-r346 RUNTIME DESTINATION bin)

if(WIN32)
  add_executable(r346-owner r346-source/r346_atlas_owner-v2.cu)
  set_target_properties(r346-owner PROPERTIES CUDA_STANDARD 17)
  target_include_directories(r346-owner PRIVATE r346-source "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/r178-headers")
  target_link_libraries(r346-owner PRIVATE r322_gmp CUDA::cudart_static CUDA::cuda_driver bcrypt)
  target_compile_options(r346-owner PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/EHsc,/Zc:preprocessor>")
endif()
